二、签到服务

JAVA 
package com.aisale.backend.service;

import com.aisale.backend.entity.CheckIn;
import com.aisale.backend.entity.PointAccount;
import com.aisale.backend.entity.PointRecord;
import com.aisale.backend.repository.CheckInRepository;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.LocalDate;
import java.util.List;

@Service
@RequiredArgsConstructor
public class CheckInService {

    private final CheckInRepository checkInRepository;
    private final PointService pointService;

    @Transactional
    public CheckIn checkIn(Long userId) {
        LocalDate today = LocalDate.now();

        // 检查今天是否已签到
        if (checkInRepository.findByUserIdAndCheckInDate(userId, today).isPresent()) {
            throw new RuntimeException("今天已经签到过了");
        }

        // 计算连续签到天数
        int continuousDays = calculateContinuousDays(userId);

        // 创建签到记录
        CheckIn checkIn = new CheckIn();
        checkIn.setUserId(userId);
        checkIn.setCheckInDate(today);
        checkIn.setContinuousDays(continuousDays);

        // 计算奖励积分
        int rewardPoints = calculateRewardPoints(continuousDays);
        checkIn.setRewardPoints(rewardPoints);

        checkIn = checkInRepository.save(checkIn);

        // 发放积分
        pointService.addPoints(userId, rewardPoints, PointRecord.PointType.CHECK_IN, 
                              checkIn.getId(), "签到奖励");

        return checkIn;
    }

    private int calculateContinuousDays(Long userId) {
        List<CheckIn> checkIns = checkInRepository.findByUserIdOrderByCheckInDateDesc(userId);
        
        if (checkIns.isEmpty()) {
            return 1;
        }

        CheckIn lastCheckIn = checkIns.get(0);
        LocalDate today = LocalDate.now();
        LocalDate yesterday = today.minusDays(1);

        // 如果最后一次签到是昨天，连续天数+1
        if (lastCheckIn.getCheckInDate().equals(yesterday)) {
            return lastCheckIn.getContinuousDays() + 1;
        }

        // 如果今天已签到，返回当前连续天数
        if (lastCheckIn.getCheckInDate().equals(today)) {
            return lastCheckIn.getContinuousDays();
        }

        // 否则重置为1
        return 1;
    }

    private int calculateRewardPoints(int continuousDays) {
        if (continuousDays >= 30) return 50;
        if (continuousDays >= 14) return 30;
        if (continuousDays >= 7) return 20;
        if (continuousDays >= 3) return 10;
        return 5;
    }

    public boolean hasCheckedInToday(Long userId) {
        return checkInRepository.findByUserIdAndCheckInDate(userId, LocalDate.now()).isPresent();
    }

    public CheckIn getTodayCheckIn(Long userId) {
        return checkInRepository.findByUserIdAndCheckInDate(userId, LocalDate.now())
                .orElse(null);
    }
}

三、积分服务

JAVA 
package com.aisale.backend.service;

import com.aisale.backend.entity.PointAccount;
import com.aisale.backend.entity.PointRecord;
import com.aisale.backend.exception.business.NotFoundException;
import com.aisale.backend.repository.PointAccountRepository;
import com.aisale.backend.repository.PointRecordRepository;
import lombok.RequiredArgsConstructor;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
@RequiredArgsConstructor
public class PointService {

    private final PointAccountRepository pointAccountRepository;
    private final PointRecordRepository pointRecordRepository;

    @Transactional
    public PointAccount getOrCreateAccount(Long userId) {
        return pointAccountRepository.findByUserId(userId)
                .orElseGet(() -> {
                    PointAccount account = new PointAccount();
                    account.setUserId(userId);
                    return pointAccountRepository.save(account);
                });
    }

    @Transactional
    public void addPoints(Long userId, int points, PointRecord.PointType type, 
                         Long relatedId, String description) {
        PointAccount account = getOrCreateAccount(userId);
        
        account.setTotalPoints(account.getTotalPoints() + points);
        account.setAvailablePoints(account.getAvailablePoints() + points);
        pointAccountRepository.save(account);

        PointRecord record = new PointRecord();
        record.setUserId(userId);
        record.setType(type);
        record.setPoints(points);
        record.setBalanceAfter(account.getAvailablePoints());
        record.setRelatedId(relatedId);
        record.setDescription(description);
        pointRecordRepository.save(record);
    }

    @Transactional
    public void deductPoints(Long userId, int points, PointRecord.PointType type,
                            Long relatedId, String description) {
        PointAccount account = getOrCreateAccount(userId);
        
        if (account.getAvailablePoints() < points) {
            throw new RuntimeException("积分不足");
        }

        account.setAvailablePoints(account.getAvailablePoints() - points);
        account.setUsedPoints(account.getUsedPoints() + points);
        pointAccountRepository.save(account);

        PointRecord record = new PointRecord();
        record.setUserId(userId);
        record.setType(type);
        record.setPoints(-points);
        record.setBalanceAfter(account.getAvailablePoints());
        record.setRelatedId(relatedId);
        record.setDescription(description);
        pointRecordRepository.save(record);
    }

    public Page<PointRecord> getPointRecords(Long userId, Pageable pageable) {
        return pointRecordRepository.findByUserId(userId, pageable);
    }
}

四、信誉服务

JAVA 
package com.aisale.backend.service;

import com.aisale.backend.entity.ReputationAccount;
import com.aisale.backend.entity.ReputationRecord;
import com.aisale.backend.entity.ReputationAccount.ReputationLevel;
import com.aisale.backend.repository.ReputationAccountRepository;
import com.aisale.backend.repository.ReputationRecordRepository;
import lombok.RequiredArgsConstructor;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
@RequiredArgsConstructor
public class ReputationService {

    private final ReputationAccountRepository reputationAccountRepository;
    private final ReputationRecordRepository reputationRecordRepository;

    @Transactional
    public ReputationAccount getOrCreateAccount(Long userId) {
        return reputationAccountRepository.findByUserId(userId)
                .orElseGet(() -> {
                    ReputationAccount account = new ReputationAccount();
                    account.setUserId(userId);
                    return reputationAccountRepository.save(account);
                });
    }

    @Transactional
    public void updateReputationAfterReview(Long userId, int rating, String content) {
        ReputationAccount account = getOrCreateAccount(userId);

        account.setTotalReviews(account.getTotalReviews() + 1);

        if (rating >= 4) {
            account.setGoodReviews(account.getGoodReviews() + 1);
            account.setTotalScore(account.getTotalScore() + 5);
        } else if (rating == 3) {
            account.setNeutralReviews(account.getNeutralReviews() + 1);
        } else {
            account.setBadReviews(account.getBadReviews() + 1);
            account.setTotalScore(account.getTotalScore() - 10);
        }

        // 更新平均分
        double totalScore = account.getGoodReviews() * 5 + 
                           account.getNeutralReviews() * 3 + 
                           account.getBadReviews() * 1;
        account.setAvgRating(account.getTotalReviews() > 0 ? 
            totalScore / account.getTotalReviews() : 0);

        // 更新等级
        ReputationLevel level = ReputationLevel.calculate(
            account.getAvgRating(), account.getTotalReviews());
        account.setLevel(level.getLevel());

        reputationAccountRepository.save(account);

        // 记录
        ReputationRecord record = new ReputationRecord();
        record.setUserId(userId);
        record.setType(ReputationRecord.RecordType.REVIEW_ADD);
        record.setScoreChange(rating >= 4 ? 5 : (rating <= 2 ? -10 : 0));
        record.setBalanceAfter(account.getTotalScore());
        record.setRating(rating);
        record.setContent(content);
        reputationRecordRepository.save(record);
    }

    public Page<ReputationRecord> getReputationRecords(Long userId, Pageable pageable) {
        return reputationRecordRepository.findByUserId(userId, pageable);
    }
}
