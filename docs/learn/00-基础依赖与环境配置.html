一、项目概述

1.1 技术栈总览

后端技术栈:

核心框架:

  • Java 17 (LTS)
  • Spring Boot 3.3.4
  • Spring Security (认证授权)
  • Spring Data JPA (数据持久化)

通信协议:

  • SSE (Server-Sent Events 实时消息)
  • WebSocket (实时通信)
  • REST API (前后端交互)

数据存储:

  • PostgreSQL (生产数据库)
  • H2 (开发环境内存数据库)
  • Redis (分布式缓存)

消息中间件:

  • RabbitMQ (消息队列)

AI集成:

  • Spring AI (OpenAI)

其他工具:

  • Lombok (代码简化)
  • JWT (Token认证 - jjwt 0.12.6)
  • Knife4j (API文档)
  • Hutool (工具库)
  • Aliyun OSS (对象存储)
  • Spring Boot DevTools (热部署)

前端技术栈:

核心框架:

  • Vue 3.5 (Composition API)
  • TypeScript 5.8
  • Vite 6.2 (构建工具)

UI组件库:

  • Element Plus 2.8

状态管理:

  • Pinia 3.0

路由管理:

  • Vue Router 5.0

HTTP客户端:

  • Axios 1.7

测试:

  • Playwright (E2E测试)

5.2 SecurityConfig 安全配置

文件位置: backend/src/main/java/com/aisale/backend/security/SecurityConfig.java

JAVA 
package com.aisale.backend.security;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.web.SecurityFilterChain;
import org.springframework.security.web.authentication.UsernamePasswordAuthenticationFilter;
import lombok.RequiredArgsConstructor;

/**
 * Spring Security配置类
 * 
 * 配置说明:
 * 1. 禁用CSRF（前后端分离架构，使用JWT）
 * 2. 禁用Session（无状态认证）
 * 3. 配置URL访问权限
 * 4. 添加JWT过滤器
 */
@Configuration
@EnableWebSecurity
@RequiredArgsConstructor
public class SecurityConfig {

    private final JwtAuthenticationFilter jwtAuthenticationFilter;

    /**
     * 配置安全过滤链
     */
    @Bean
    public SecurityFilterChain filterChain(HttpSecurity http) throws Exception {
        http
            // 禁用CSRF（前后端分离）
            .csrf(csrf -> csrf.disable())

            // 禁用Session（JWT无状态）
            .sessionManagement(session -> 
                session.sessionCreationPolicy(SessionCreationPolicy.STATELESS))

            // 配置URL访问权限
            .authorizeHttpRequests(auth -> auth
                // 公开接口
                .requestMatchers(
                    "/api/auth/**",
                    "/api/user/products/public/**",
                    "/h2-console/**",
                    "/swagger-ui/**",
                    "/v3/api-docs/**",
                    "/doc.html"
                ).permitAll()

                // Admin接口需要ADMIN角色
                .requestMatchers("/api/admin/**").hasAnyRole("ADMIN", "SUPER_ADMIN")

                // 其他接口需要认证
                .anyRequest().authenticated()
            )

            // 添加JWT过滤器
            .addFilterBefore(
                jwtAuthenticationFilter,
                UsernamePasswordAuthenticationFilter.class
            );

        return http.build();
    }

    /**
     * 密码加密器
     */
    @Bean
    public PasswordEncoder passwordEncoder() {
        return new BCryptPasswordEncoder();
    }
}

5.3 WebConfig 跨域配置

文件位置: backend/src/main/java/com/aisale/backend/config/WebConfig.java

JAVA 
package com.aisale.backend.config;

import org.springframework.context.annotation.Configuration;
import org.springframework.web.servlet.config.annotation.CorsRegistry;
import org.springframework.web.servlet.config.annotation.WebMvcConfigurer;

/**
 * Web配置
 * 
 * 配置说明:
 * 1. 配置CORS跨域
 * 2. 允许前端开发服务器访问
 */
@Configuration
public class WebConfig implements WebMvcConfigurer {

    @Override
    public void addCorsMappings(CorsRegistry registry) {
        registry.addMapping("/**")
            .allowedOriginPatterns(
                "http://localhost:*",
                "http://127.0.0.1:*"
            )
            .allowedMethods("GET", "POST", "PUT", "DELETE", "OPTIONS")
            .allowedHeaders("*")
            .allowCredentials(true)
            .maxAge(3600);
    }
}

import org.springframework.data.redis.core.RedisTemplate;
import org.springframework.data.redis.serializer.GenericJackson2JsonRedisSerializer;
import org.springframework.data.redis.serializer.StringRedisSerializer;

/**
 * Redis配置类
 * 
 * 配置说明:
 * 1. 配置RedisTemplate
 * 2. 使用JSON序列化
 */
@Configuration
public class RedisConfig {

    @Bean
    public RedisTemplate<String, Object> redisTemplate(RedisConnectionFactory connectionFactory) {
        RedisTemplate<String, Object> template = new RedisTemplate<>();
        template.setConnectionFactory(connectionFactory);
        
        // 使用String序列化器处理key
        template.setKeySerializer(new StringRedisSerializer());
        // 使用JSON序列化器处理value
        template.setValueSerializer(new GenericJackson2JsonRedisSerializer());
        
        template.afterPropertiesSet();
        return template;
    }
}

JAVA 
package com.aisale.backend.config;

import com.aliyun.oss.OSS;
import com.aliyun.oss.OSSClientBuilder;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import lombok.RequiredArgsConstructor;

/**
 * 阿里云OSS配置类
 * 
 * 配置说明:
 * 1. 配置OSS客户端
 * 2. 提供文件上传能力
 */
@Configuration
@RequiredArgsConstructor
public class OssConfig {

    private final OssProperties ossProperties;

    @Bean
    public OSS ossClient() {
        return new OSSClientBuilder()
            .build(
                ossProperties.getEndpoint(),
                ossProperties.getAccessKeyId(),
                ossProperties.getAccessKeySecret()
            );
    }
}
