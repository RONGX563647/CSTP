10 商品搜索与推荐模块

学习时长: 1-2 天

难度: ⭐⭐⭐☆☆

前置知识: 全文搜索、相关性算法、缓存机制

---

一、本章概述

本章实现商品搜索和推荐功能，包含：

  • 关键词搜索：支持名称、描述、标签、分类多字段搜索
  • 同义词扩展：自动扩展搜索词（如"手机"扩展为"iphone,华为,小米"等）
  • 相关性排序：综合名称匹配、热度、折扣等多维度评分
  • 热门搜索：统计热门关键词
  • 搜索历史：记录用户搜索行为

二、ProductSearchService 搜索服务

2.1 技术说明

ProductSearchService 实现智能搜索功能，核心算法包括：

1. 同义词扩展

JAVA 
SYNONYM_MAP.put("手机", Arrays.asList("电话", "移动", "iphone", "华为", "小米"));
SYNONYM_MAP.put("电脑", Arrays.asList("笔记本", "计算机", "pc", "macbook"));

2. 相关性评分算法

  • 名称完全匹配: +200分
  • 名称前缀匹配: +150分
  • 名称包含匹配: +120分（位置越靠前分数越高）
  • 分类匹配: +60-80分
  • 标签匹配: +45-60分
  • 描述匹配: +25分/次（最多5次）
  • 同义词匹配: +15-35分
  • 销量加权: +0.5分/销量
  • 浏览量加权: +0.05分/浏览
  • 推荐商品: +30分
  • 折扣商品: +20分

package com.aisale.backend.service;

import com.aisale.backend.dto.ProductResponse;
import com.aisale.backend.dto.ProductSearchResult;
import com.aisale.backend.entity.Product;
import com.aisale.backend.repository.ProductRepository;
import com.aisale.backend.repository.SearchHistoryRepository;
import lombok.RequiredArgsConstructor;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.PageRequest;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.math.BigDecimal;
import java.util.*;
import java.util.stream.Collectors;

@Service
@RequiredArgsConstructor
public class ProductSearchService {

    private final ProductRepository productRepository;
    private final SearchHistoryRepository searchHistoryRepository;

    /**
     * 同义词映射表
     */
    private static final Map<String, List<String>> SYNONYM_MAP = new HashMap<>();

    static {
        SYNONYM_MAP.put("手机", Arrays.asList("电话", "移动", "iphone", "华为", "小米"));
        SYNONYM_MAP.put("电脑", Arrays.asList("笔记本", "计算机", "pc", "macbook"));
        SYNONYM_MAP.put("平板", Arrays.asList("ipad", "tablet", "pad"));
        SYNONYM_MAP.put("耳机", Arrays.asList("earphone", "headphone", "airpods"));
    }

    /**
     * 搜索商品（带相关性排序）
     */
    @Transactional
    public Page<ProductSearchResult> searchWithRelevance(
            String keyword, String category,
            BigDecimal minPrice, BigDecimal maxPrice,
            Long userId, int page, int size) {
        
        // 记录搜索历史
        if (keyword != null && !keyword.trim().isEmpty()) {
            recordSearchHistory(userId, keyword.trim());
        }

        // 1. 用原始关键词搜索
        Page<Product> products = productRepository.searchMultiField(
            keyword, category, minPrice, maxPrice, Product.ProductStatus.ON_SALE, 
            PageRequest.of(page, size)
        );

        List<Product> allProducts = new ArrayList<>(products.getContent());

        // 2. 如果结果不足，使用同义词扩展搜索
        if (keyword != null && allProducts.size() < size) {
            List<String> synonyms = expandKeywords(keyword.trim());
            for (String synonym : synonyms) {
                Page<Product> synonymResults = productRepository.searchMultiField(
                    synonym, category, minPrice, maxPrice, 
                    Product.ProductStatus.ON_SALE, PageRequest.of(page, size)
                );
                allProducts.addAll(synonymResults.getContent());
            }
        }

        // 3. 计算相关性得分并排序
        List<ProductSearchResult> results = allProducts.stream()
            .map(product -> {
                double score = calculateEnhancedRelevance(product, keyword);
                return ProductSearchResult.builder()
                    .product(ProductResponse.fromEntity(product))
                    .relevanceScore(score)
                    .matchType(determineMatchType(product, keyword))
                    .build();
            })
            .sorted(Comparator.comparingDouble(ProductSearchResult::getRelevanceScore).reversed())
            .collect(Collectors.toList());

        return new PageImpl<>(results, PageRequest.of(page, size), results.size());
    }

    /**
     * 计算相关性得分
     */
    private double calculateEnhancedRelevance(Product product, String keyword) {
        double score = 0.0;
        String lowerKeyword = keyword.toLowerCase().trim();

        // 名称匹配（权重最高）
        if (product.getName() != null) {
            String lowerName = product.getName().toLowerCase();
            if (lowerName.equals(lowerKeyword)) {
                score += 200.0; // 完全匹配
            } else if (lowerName.startsWith(lowerKeyword)) {
                score += 150.0; // 前缀匹配
            } else if (lowerName.contains(lowerKeyword)) {
                score += 120.0; // 包含匹配
            }
        }

        // 分类匹配
        if (product.getCategory() != null && product.getCategory().toLowerCase().contains(lowerKeyword)) {
            score += 60.0;
        }

        // 标签匹配
        if (product.getTags() != null) {
            for (String tag : product.getTags()) {
                if (tag.toLowerCase().contains(lowerKeyword)) {
                    score += 45.0;
                }
            }
        }

        // 描述匹配
        if (product.getDescription() != null && product.getDescription().toLowerCase().contains(lowerKeyword)) {
            score += 25.0;
        }

        // 热度加权
        score += product.getSalesCount() * 0.5;
        score += product.getViewCount() * 0.05;

        // 推荐商品加分
        if (product.getIsFeatured() != null && product.getIsFeatured()) {
            score += 30.0;
        }

        return score;
    }

    /**
     * 获取热门商品
     */
    @Transactional(readOnly = true)
    public List<ProductResponse> getHotProducts(int limit) {
        return productRepository.findTopBySalesCount(
            Product.ProductStatus.ON_SALE, PageRequest.of(0, limit)
        ).stream().map(ProductResponse::fromEntity).collect(Collectors.toList());
    }

JAVA 
package com.aisale.backend.entity;

import jakarta.persistence.*;
import lombok.Data;
import java.time.LocalDateTime;

@Data
@Entity
@Table(name = "search_histories", indexes = {
    @Index(name = "idx_search_user", columnList = "user_id"),
    @Index(name = "idx_search_keyword", columnList = "keyword")
})
public class SearchHistory {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Column(name = "user_id", nullable = false)
    private Long userId;

    @Column(nullable = false, length = 100)
    private String keyword;

    @Column(name = "search_count")
    private Integer searchCount = 1;

    @Column(name = "last_search_time")
    private LocalDateTime lastSearchTime;

    @PrePersist
    protected void onCreate() {
        lastSearchTime = LocalDateTime.now();
    }
}
