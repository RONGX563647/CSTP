二、Python API测试

文件位置: backend/tests/apifullcoverage_test.py

PYTHON 
import requests
import json
import sys

BASE_URL = "http://localhost:8090/api"
TOKEN = None
ADMIN_TOKEN = None

def test_auth():
    """测试认证模块"""
    print("\n=== 测试认证模块 ===")
    
    # 登录
    response = requests.post(f"{BASE_URL}/auth/login", json={
        "username": "zhangsan",
        "password": "123456"
    })
    assert response.status_code == 200
    data = response.json()
    assert data['code'] == 200
    TOKEN = data['data']['token']
    print(f"✓ 用户登录成功")
    
    # 管理员登录
    response = requests.post(f"{BASE_URL}/admin/auth/login", json={
        "username": "admin",
        "password": "123456"
    })
    assert response.status_code == 200
    data = response.json()
    ADMIN_TOKEN = data['data']['token']
    print(f"✓ 管理员登录成功")

def test_products():
    """测试商品模块"""
    print("\n=== 测试商品模块 ===")
    
    # 获取公开商品列表
    response = requests.get(f"{BASE_URL}/user/products/public/on-sale")
    assert response.status_code == 200
    print(f"✓ 获取商品列表成功")
    
    # 创建商品
    headers = {"Authorization": f"Bearer {TOKEN}"}
    response = requests.post(f"{BASE_URL}/user/products", 
                            headers=headers,
                            json={
                                "name": "测试商品",
                                "description": "这是一个测试商品",
                                "price": 99.99,
                                "originalPrice": 199.99,
                                "stock": 10,
                                "category": "测试"
                            })
    assert response.status_code == 200
    print(f"✓ 创建商品成功")

def test_orders():
    """测试订单模块"""
    print("\n=== 测试订单模块 ===")
    
    headers = {"Authorization": f"Bearer {TOKEN}"}
    
    # 获取订单列表
    response = requests.get(f"{BASE_URL}/user/orders/buyer", headers=headers)
    assert response.status_code == 200
    print(f"✓ 获取订单列表成功")

def run_all_tests():
    """运行所有测试"""
    try:
        test_auth()
        test_products()
        test_orders()
        print("\n✅ 所有测试通过")
    except Exception as e:
        print(f"\n❌ 测试失败: {e}")
        sys.exit(1)

if __name__ == "__main__":
    run_all_tests()

运行测试:

BASH 
cd backend/tests
python api_full_coverage_test.py

三、Docker部署

3.1 Docker Compose配置

文件位置: docker-compose.yml

YAML 
version: '3.8'

services:
  backend:
    build:
      context: ./backend
      dockerfile: Dockerfile
    ports:
      - "8090:8090"
    environment:
      - SPRING_PROFILES_ACTIVE=prod
      - DB_HOST=postgres
      - DB_PORT=5432
      - DB_NAME=aisale
      - DB_USERNAME=aisale
      - DB_PASSWORD=aisale123
      - REDIS_HOST=redis
      - REDIS_PORT=6379
    depends_on:
      - postgres
      - redis
    networks:
      - aisale-network

  frontend:
    build:
      context: ./frontend
      dockerfile: Dockerfile
    ports:
      - "80:80"
    depends_on:
      - backend
    networks:
      - aisale-network

  postgres:
    image: postgres:15
    environment:
      - POSTGRES_DB=aisale
      - POSTGRES_USER=aisale
      - POSTGRES_PASSWORD=aisale123
    volumes:
      - postgres-data:/var/lib/postgresql/data
    ports:
      - "5432:5432"
    networks:
      - aisale-network

  redis:
    image: redis:7
    ports:
      - "6379:6379"
    networks:
      - aisale-network

volumes:
  postgres-data:

networks:
  aisale-network:
    driver: bridge

3.2 后端Dockerfile

文件位置: backend/Dockerfile

DOCKERFILE 
FROM maven:3.9-eclipse-temurin-17 AS build
WORKDIR /app
COPY pom.xml .
RUN mvn dependency:go-offline
COPY src ./src
RUN mvn clean package -DskipTests

FROM eclipse-temurin:17-jre
WORKDIR /app
COPY --from=build /app/target/*.jar app.jar
EXPOSE 8090
ENTRYPOINT ["java", "-jar", "app.jar"]

3.3 前端Dockerfile

文件位置: frontend/Dockerfile

DOCKERFILE 
FROM node:20 AS build
WORKDIR /app
COPY package*.json ./
RUN npm install
COPY . .
RUN npm run build

FROM nginx:alpine
COPY --from=build /app/dist /usr/share/nginx/html
COPY nginx.conf /etc/nginx/conf.d/default.conf
EXPOSE 80
CMD ["nginx", "-g", "daemon off;"]

3.4 Nginx配置

文件位置: frontend/nginx.conf

NGINX 
server {
    listen 80;
    server_name localhost;

    location / {
        root /usr/share/nginx/html;
        index index.html;
        try_files $uri $uri/ /index.html;
    }

    location /api/ {
        proxy_pass http://backend:8090/api/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }
}

四、部署流程

4.1 本地开发

BASH 
# 1. 启动后端
cd backend
mvn spring-boot:run -Dspring-boot.run.profiles=dev

# 2. 启动前端
cd frontend
npm install
npm run dev

4.2 Docker部署

BASH 
# 1. 构建并启动所有服务
docker-compose up -d

# 2. 查看服务状态
docker-compose ps

# 3. 查看日志
docker-compose logs -f backend

# 4. 停止服务
docker-compose down

4.3 生产部署

BASH 
# 1. 使用生产配置
docker-compose -f docker-compose.prod.yml up -d

# 2. 检查健康状态
curl http://localhost:8090/actuator/health

# 3. 备份数据库
docker exec -t postgres pg_dump -U aisale aisale > backup.sql
